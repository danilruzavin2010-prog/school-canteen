import sqlite3
import datetime

conn = sqlite3.connect("canteen.db")
cur = conn.cursor()

cur.execute('''
CREATE TABLE IF NOT EXISTS users (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    vk_id INTEGER UNIQUE,
    full_name TEXT,
    department TEXT
)
''')

cur.execute('''
CREATE TABLE IF NOT EXISTS menu (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    date TEXT,
    meal_type TEXT,
    dish_name TEXT,
    price REAL
)
''')

cur.execute('''
CREATE TABLE IF NOT EXISTS orders (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    user_id INTEGER,
    menu_id INTEGER,
    order_date TEXT,
    status TEXT DEFAULT 'новый'
)
''')

today = datetime.date.today().isoformat()
menu_items = [
    (today, 'завтрак', 'Каша овсяная', 90.0),
    (today, 'завтрак', 'Яичница', 75.0),
    (today, 'обед', 'Суп куриный', 120.0),
    (today, 'обед', 'Гречка с мясом', 150.0),
    (today, 'ужин', 'Рыба с рисом', 130.0),
]
cur.executemany("INSERT OR IGNORE INTO menu (date, meal_type, dish_name, price) VALUES (?, ?, ?, ?)", menu_items)

conn.commit()
conn.close()
print("База создана и меню загружено")
