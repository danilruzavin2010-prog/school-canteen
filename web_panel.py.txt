from flask import Flask, render_template_string, request
import sqlite3
import datetime

app = Flask(__name__)
DB = "canteen.db"

HTML = """
<!DOCTYPE html>
<html>
<head><title>Панель столовой</title></head>
<body>
<h2>Заказы на {{ date }}</h2>
<table border="1">
<tr><th>ФИО</th><th>Отдел</th><th>Приём</th><th>Блюдо</th><th>Цена</th><th>Статус</th></tr>
{% for row in orders %}
<tr>
  <td>{{ row[0] }}</td>
  <td>{{ row[1] }}</td>
  <td>{{ row[2] }}</td>
  <td>{{ row[3] }}</td>
  <td>{{ row[4] }}</td>
  <td>{{ row[5] }}</td>
</tr>
{% endfor %}
</table>
<p>Итого: {{ total }} руб., порций: {{ count }}</p>
<form method="GET">
  <input type="date" name="date" value="{{ date }}">
  <button>Показать</button>
</form>
</body>
</html>
"""

@app.route('/')
def panel():
    date_str = request.args.get('date', datetime.date.today().isoformat())
    conn = sqlite3.connect(DB)
    cur = conn.cursor()
    cur.execute('''
        SELECT u.full_name, u.department, m.meal_type, m.dish_name, m.price, o.status
        FROM orders o
        JOIN users u ON o.user_id = u.id
        JOIN menu m ON o.menu_id = m.id
        WHERE DATE(o.order_date) = ?
    ''', (date_str,))
    rows = cur.fetchall()
    conn.close()
    total = sum(r[4] for r in rows) if rows else 0
    return render_template_string(HTML, orders=rows, total=total, count=len(rows), date=date_str)

if __name__ == '__main__':
    app.run(host='0.0.0.0', port=5000, debug=False)
