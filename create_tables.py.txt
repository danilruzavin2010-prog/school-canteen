import os
import urllib.parse
import psycopg2
import sys

def get_db_connection():
    db_url = os.environ.get("DATABASE_URL")
    if not db_url:
        print("❌ Переменная DATABASE_URL не найдена!")
        print("Убедись, что она добавлена в разделе 'Переменные' в RelaxDev")
        sys.exit(1)
    
    print(f"🔗 Подключение к БД...")
    result = urllib.parse.urlparse(db_url)
    try:
        conn = psycopg2.connect(
            database=result.path[1:],
            user=result.username,
            password=result.password,
            host=result.hostname,
            port=result.port
        )
        print("✅ Подключение успешно")
        return conn
    except Exception as e:
        print(f"❌ Ошибка подключения: {e}")
        sys.exit(1)

def create_tables():
    conn = get_db_connection()
    cur = conn.cursor()
    
    print("🔄 Создание таблиц...")
    
    # Таблица users
    cur.execute('''
        CREATE TABLE IF NOT EXISTS users (
            id SERIAL PRIMARY KEY,
            vk_id BIGINT UNIQUE,
            full_name TEXT,
            department TEXT
        );
    ''')
    
    # Таблица orders
    cur.execute('''
        CREATE TABLE IF NOT EXISTS orders (
            id SERIAL PRIMARY KEY,
            user_id INTEGER REFERENCES users(id),
            class_name TEXT,
            order_date DATE,
            meal_type TEXT,
            count_plat INTEGER DEFAULT 0,
            count_bes INTEGER DEFAULT 0,
            count_svo INTEGER DEFAULT 0,
            count_ovz INTEGER DEFAULT 0,
            status TEXT DEFAULT 'новый'
        );
    ''')
    
    conn.commit()
    conn.close()
    print("✅ Таблицы users и orders успешно созданы!")

if __name__ == "__main__":
    create_tables()
