DB_NAME = "canteen.db"

def init_db():
    conn = sqlite3.connect(DB_NAME)
    cur = conn.cursor()
    cur.executescript('''
        CREATE TABLE IF NOT EXISTS users (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            vk_id INTEGER UNIQUE,
            full_name TEXT,
            department TEXT
        );
        CREATE TABLE IF NOT EXISTS menu (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            date TEXT,
            meal_type TEXT,
            dish_name TEXT,
            price REAL
        );
        CREATE TABLE IF NOT EXISTS orders (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            user_id INTEGER,
            menu_id INTEGER,
            order_date TEXT,
            status TEXT DEFAULT 'новый'
        );
    ''')
    conn.commit()
    conn.close()

def add_user(vk_id, name, dept="не указан"):
    conn = sqlite3.connect(DB_NAME)
    cur = conn.cursor()
    cur.execute("INSERT OR IGNORE INTO users (vk_id, full_name, department) VALUES (?, ?, ?)", (vk_id, name, dept))
    conn.commit()
    conn.close()

def get_user_id(vk_id):
    conn = sqlite3.connect(DB_NAME)
    cur = conn.cursor()
    cur.execute("SELECT id FROM users WHERE vk_id = ?", (vk_id,))
    row = cur.fetchone()
    conn.close()
    return row[0] if row else None

def get_today_menu():
    today = datetime.date.today().isoformat()
    conn = sqlite3.connect(DB_NAME)
    cur = conn.cursor()
    cur.execute("SELECT id, meal_type, dish_name, price FROM menu WHERE date = ?", (today,))
    rows = cur.fetchall()
    conn.close()
    return rows

def create_order(user_id, menu_id):
    conn = sqlite3.connect(DB_NAME)
    cur = conn.cursor()
    cur.execute("INSERT INTO orders (user_id, menu_id, order_date) VALUES (?, ?, ?)",
                (user_id, menu_id, datetime.datetime.now().isoformat()))
    conn.commit()
    conn.close()

def get_orders_for_staff(date_filter=None):
    conn = sqlite3.connect(DB_NAME)
    cur = conn.cursor()
    if date_filter:
        cur.execute('''
            SELECT u.full_name, u.department, m.meal_type, m.dish_name, m.price, o.status, o.order_date
            FROM orders o
            JOIN users u ON o.user_id = u.id
            JOIN menu m ON o.menu_id = m.id
            WHERE DATE(o.order_date) = ?
        ''', (date_filter,))
    else:
        cur.execute('''
            SELECT u.full_name, u.department, m.meal_type, m.dish_name, m.price, o.status, o.order_date
            FROM orders o
            JOIN users u ON o.user_id = u.id
            JOIN menu m ON o.menu_id = m.id
            WHERE DATE(o.order_date) = DATE('now')
        ''')
    return cur.fetchall()

def handle_message(event, vk):
    msg = event.obj.message['text'].lower().strip()
    from_id = event.obj.message['from_id']
    user_info = vk.method('users.get', {'user_ids': from_id})
    name = f"{user_info[0]['first_name']} {user_info[0]['last_name']}"
    add_user(from_id, name)

    if msg.startswith("!стафф"):
        staff_ids = [523723395]  
        if from_id not in staff_ids:
            vk.method('messages.send', {
                'user_id': from_id,
                'message': "Доступ запрещён.",
                'random_id': 0
            })
            return
        parts = msg.split()
        if len(parts) > 1 and parts[1] == "сегодня":
            rows = get_orders_for_staff(datetime.date.today().isoformat())
        else:
            rows = get_orders_for_staff()
        if not rows:
            reply = "Заказов на сегодня нет."
        else:
            reply = "📋 ЗАКАЗЫ НА СЕГОДНЯ:\n\n"
            total = 0
            for row in rows:
                reply += f"{row[0]} ({row[1]}) – {row[2]}: {row[3]} ({row[4]} руб.) – статус: {row[5]}\n"
                total += row[4]
            reply += f"\nИтого порций: {len(rows)}, общая сумма: {total} руб."
        vk.method('messages.send', {
            'user_id': from_id,
            'message': reply,
            'random_id': 0
        })
        return

    if msg.startswith("заказать"):
        today_menu = get_today_menu()
        if not today_menu:
            vk.method('messages.send', {
                'user_id': from_id,
                'message': "Меню на сегодня не загружено.",
                'random_id': 0
            })
            return
        now_hour = datetime.datetime.now().hour
        if now_hour >= ORDER_DEADLINE_HOUR:
            vk.method('messages.send', {
                'user_id': from_id,
                'message': f"Заказы принимаются до {ORDER_DEADLINE_HOUR}:00.",
                'random_id': 0
            })
            return
        reply = "🍽 МЕНЮ НА СЕГОДНЯ:\n"
        for idx, item in enumerate(today_menu, 1):
            reply += f"{idx}. {item[1]} – {item[2]} ({item[3]} руб.)\n"
        reply += "\nНапишите номер блюда."
        vk.method('messages.send', {
            'user_id': from_id,
            'message': reply,
            'random_id': 0
        })
    elif msg.isdigit() and len(msg) <= 2:
        user_db_id = get_user_id(from_id)
        if not user_db_id:
            vk.method('messages.send', {
                'user_id': from_id,
                'message': "Сначала напишите 'заказать'.",
                'random_id': 0
            })
            return
        menu_items = get_today_menu()
        choice = int(msg)
        if 1 <= choice <= len(menu_items):
            menu_id = menu_items[choice-1][0]
            create_order(user_db_id, menu_id)
            vk.method('messages.send', {
                'user_id': from_id,
                'message': f"✅ Заказ оформлен: {menu_items[choice-1][1]} – {menu_items[choice-1][2]}.",
                'random_id': 0
            })
        else:
            vk.method('messages.send', {
                'user_id': from_id,
                'message': "Некорректный номер.",
                'random_id': 0
            })
