import threading
import os
import time
import subprocess

def run_bot():
    """Запускает бота и следит за ним — если упадёт, перезапустит"""
    while True:
        try:
            print("🚀 Запуск бота...")
            subprocess.run(["python", "bot.py"])
        except Exception as e:
            print(f"❌ Бот упал: {e}")
        print("⚠️ Бот завершился. Перезапуск через 5 секунд...")
        time.sleep(5)

if __name__ == "__main__":
    bot_thread = threading.Thread(target=run_bot, daemon=True)
    bot_thread.start()
    time.sleep(2)
    os.system("gunicorn web_panel:app --bind 0.0.0.0:5000")
